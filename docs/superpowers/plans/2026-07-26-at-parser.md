# AT Local Parser Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

> **Implementation discipline:** Load `superpowers:test-driven-development` before Tasks 1-6, and load `browser:control-in-app-browser` before the browser verification in Task 7.

**Goal:** Build a single-file, offline Chinese web page that decodes one JWT-shaped ChatGPT access token locally, explains its claims, masks sensitive values, and never validates or transmits the token.

**Architecture:** The runtime application is one `index.html` containing all HTML, CSS, and JavaScript. Pure parsing, time interpretation, redaction, and grouping functions are exported through `globalThis.ATParserCore` so Node's built-in test runner can load the inline script without browser dependencies; browser-only boot code runs only when `document` exists.

**Tech Stack:** HTML5, CSS, vanilla JavaScript, Node.js 24 built-in `node:test`, `node:vm`, and the in-app browser for final interaction and responsive verification.

---

## File map

- Create `index.html`: the entire user-facing application, inline styles, pure core functions, and browser controller.
- Create `tests/helpers/load-core.mjs`: extracts the application script from `index.html` and evaluates it in an isolated Node VM.
- Create `tests/parser.test.mjs`: normalization, JWT decoding, time status, and warning tests.
- Create `tests/redaction.test.mjs`: recursive masking and field-category tests.
- Create `tests/page.test.mjs`: static CSP, privacy, accessibility, and page-structure tests.
- Create `tests/interactions.test.mjs`: reveal timer and clipboard-path tests.
- Create `tests/readme.test.mjs`: verifies the operating and security guidance.
- Create `tests/repo-safety.test.mjs`: prevents committing token-shaped secrets.
- Create `README.md`: local-use instructions, limitations, and contributor verification commands.

The shipped application remains a single file. Files under `tests/` and `README.md` are development support only and are not required to run the parser.

```js
import { Buffer } from "node:buffer";
import { readFileSync } from "node:fs";
import vm from "node:vm";

export function loadCore() {
  const htmlUrl = new URL("../../index.html", import.meta.url);
  const html = readFileSync(htmlUrl, "utf8");
  const match = html.match(/<script id="app-script">([\s\S]*?)<\/script>/u);

  if (!match) {
    throw new Error("index.html is missing <script id=\"app-script\">");
  }

  const context = {
    Date,
    Intl,
    TextDecoder,
    TextEncoder,
    URLSearchParams,
    clearTimeout,
    console,
    setTimeout,
    atob(value) {
      return Buffer.from(value, "base64").toString("binary");
    },
  };

  context.globalThis = context;
  vm.createContext(context);
  vm.runInContext(match[1], context, { filename: "index.html#app-script" });

  if (!context.ATParserCore) {
    throw new Error("app script did not expose globalThis.ATParserCore");
  }

  return context.ATParserCore;
}

export function makeJwt(header, payload, signature = "synthetic-signature") {
  const encode = (value) => Buffer.from(JSON.stringify(value), "utf8").toString("base64url");
  return [encode(header), encode(payload), signature].join(".");
}

export function toPlain(value) {
  return JSON.parse(JSON.stringify(value));
}
```

```html
<!doctype html>
<html lang="zh-CN">
<head>
  <meta charset="utf-8">
  <meta name="viewport" content="width=device-width, initial-scale=1">
  <meta name="color-scheme" content="light dark">
  <meta http-equiv="Content-Security-Policy" content="default-src 'none'; script-src 'unsafe-inline'; style-src 'unsafe-inline'; connect-src 'none'; img-src data:; font-src 'none'; frame-src 'none'; form-action 'none'; base-uri 'none'">
  <title>AT 本地解析器</title>
</head>
<body>
  <main>
    <h1>AT 本地解析器</h1>
  </main>
  <script id="app-script">
  (() => {
    "use strict";

    class ParserError extends Error {
      constructor(code, message) {
        super(message);
        this.name = "ParserError";
        this.code = code;
      }
    }

    function normalizeInput(input) {
      return String(input ?? "")
        .trim()
        .replace(/^Bearer\s+/iu, "")
        .trim();
    }

    function decodeJsonObject(segment, label) {
      if (!/^[A-Za-z0-9_-]+={0,2}$/u.test(segment)) {
        throw new ParserError(`${label.toUpperCase()}_BASE64URL`, `${label} 不是合法的 Base64URL。`);
      }

      const unpadded = segment.replace(/=+$/u, "");
      if (unpadded.length % 4 === 1) {
        throw new ParserError(`${label.toUpperCase()}_BASE64URL`, `${label} 不是合法的 Base64URL。`);
      }

      const standard = unpadded.replace(/-/gu, "+").replace(/_/gu, "/");
      const padded = standard.padEnd(Math.ceil(standard.length / 4) * 4, "=");
      let binary;

      try {
        binary = atob(padded);
      } catch {
        throw new ParserError(`${label.toUpperCase()}_BASE64URL`, `${label} 不是合法的 Base64URL。`);
      }

      const bytes = Uint8Array.from(binary, (character) => character.charCodeAt(0));
      let text;

      try {
        text = new TextDecoder("utf-8", { fatal: true }).decode(bytes);
      } catch {
        throw new ParserError(`${label.toUpperCase()}_UTF8`, `${label} 不是合法的 UTF-8 文本。`);
      }

      let value;
      try {
        value = JSON.parse(text);
      } catch {
        throw new ParserError(`${label.toUpperCase()}_JSON`, `${label} 不是合法的 JSON。`);
      }

      if (value === null || Array.isArray(value) || typeof value !== "object") {
        throw new ParserError(`${label.toUpperCase()}_NOT_OBJECT`, `${label} 顶层必须是 JSON 对象。`);
      }

      return value;
    }

    function parseJwt(input) {
      const normalized = normalizeInput(input);
      if (!normalized) {
        throw new ParserError("EMPTY_INPUT", "请输入一个 AT。数据只会在当前页面内解析。");
      }

      const parts = normalized.split(".");
      if (parts.length !== 3 || parts.some((part) => part.length === 0)) {
        throw new ParserError("JWT_STRUCTURE", "AT 必须是由两个点分隔的三段式 JWT。");
      }

      return {
        header: decodeJsonObject(parts[0], "Header"),
        payload: decodeJsonObject(parts[1], "Payload"),
        signature: { present: true, verified: false },
      };
    }

    globalThis.ATParserCore = Object.freeze({
      ParserError,
      decodeJsonObject,
      normalizeInput,
      parseJwt,
    });
  })();
  </script>
</body>
</html>
```

```js
    const COMMON_SIGNING_ALGORITHMS = new Set([
      "RS256", "RS384", "RS512",
      "PS256", "PS384", "PS512",
      "ES256", "ES384", "ES512",
      "EdDSA",
      "HS256", "HS384", "HS512",
    ]);

    function parseNumericDate(payload, key) {
      if (!Object.prototype.hasOwnProperty.call(payload, key)) {
        return { key, present: false, valid: false, raw: undefined };
      }

      const raw = payload[key];
      if (typeof raw !== "number" || !Number.isFinite(raw)) {
        return { key, present: true, valid: false, raw };
      }

      const milliseconds = raw * 1000;
      const beijing = formatBeijingTime(raw);
      if (!Number.isFinite(milliseconds) || beijing === null) {
        return { key, present: true, valid: false, raw };
      }

      return {
        key,
        present: true,
        valid: true,
        raw,
        milliseconds,
        beijing,
      };
    }

    function formatBeijingTime(seconds) {
      if (typeof seconds !== "number" || !Number.isFinite(seconds)) {
        return null;
      }

      const shifted = new Date(seconds * 1000 + 8 * 60 * 60 * 1000);
      if (Number.isNaN(shifted.getTime())) {
        return null;
      }

      return `${shifted.toISOString().slice(0, 19).replace("T", " ")} +08:00`;
    }

    function evaluateTimeStatus(payload, nowMilliseconds = Date.now()) {
      const claims = {
        iat: parseNumericDate(payload, "iat"),
        nbf: parseNumericDate(payload, "nbf"),
        exp: parseNumericDate(payload, "exp"),
      };
      const invalidClaims = Object.values(claims)
        .filter((claim) => claim.present && !claim.valid)
        .map((claim) => claim.key);

      if (claims.nbf.valid && nowMilliseconds < claims.nbf.milliseconds) {
        return { code: "not_yet_valid", label: "尚未生效", claims, invalidClaims };
      }
      if (claims.exp.valid && nowMilliseconds >= claims.exp.milliseconds) {
        return { code: "expired", label: "已过期", claims, invalidClaims };
      }
      if (claims.nbf.valid || claims.exp.valid) {
        return { code: "within_window", label: "在声明时间窗口内", claims, invalidClaims };
      }
      return { code: "missing_time", label: "缺少时间声明", claims, invalidClaims };
    }

    function buildWarnings(header, time) {
      const warnings = [{
        code: "SIGNATURE_UNVERIFIED",
        message: "只完成了本地解码，未验证签名、撤销状态或服务器可用性。",
      }];
      const algorithm = typeof header.alg === "string" ? header.alg : "";

      if (!algorithm) {
        warnings.push({ code: "MISSING_ALG", message: "Header 缺少 alg 字段。" });
      } else if (algorithm.toLowerCase() === "none") {
        warnings.push({ code: "ALG_NONE", message: "alg=none 不提供签名保护。" });
      } else if (!COMMON_SIGNING_ALGORITHMS.has(algorithm)) {
        warnings.push({ code: "UNKNOWN_ALG", message: `算法 ${algorithm} 不在常见签名算法列表中。` });
      }

      for (const key of time.invalidClaims) {
        warnings.push({
          code: "INVALID_TIME_CLAIM",
          message: `${key} 不是有效的 JWT NumericDate 数字。`,
        });
      }

      if (time.code === "expired") {
        warnings.push({ code: "TOKEN_EXPIRED", message: "根据本机时间，该 token 已经过期。" });
      } else if (time.code === "not_yet_valid") {
        warnings.push({ code: "TOKEN_NOT_YET_VALID", message: "根据本机时间，该 token 尚未生效。" });
      } else if (time.code === "missing_time") {
        warnings.push({ code: "MISSING_TIME", message: "Payload 缺少有效的 nbf 和 exp 时间声明。" });
      }

      return warnings;
    }
```

Add these definitions before `parseJwt`:

```js
    const SECTION_DEFINITIONS = [
      { id: "identity", title: "账号与身份" },
      { id: "authentication", title: "认证信息" },
      { id: "permissions", title: "权限" },
      { id: "time", title: "时间" },
      { id: "security", title: "安全信息" },
      { id: "other", title: "其他字段" },
    ];

    function isSensitiveKey(key) {
      const normalized = String(key).toLowerCase();
      if (["email", "name", "sub", "jti", "session_id"].includes(normalized)) {
        return true;
      }
      return /(?:^|_)(?:id|ids)$/u.test(normalized);
    }

    function redactDeep(value, key = "") {
      if (isSensitiveKey(key)) {
        if (Array.isArray(value)) {
          return value.map(() => "[REDACTED]");
        }
        return "[REDACTED]";
      }

      if (Array.isArray(value)) {
        return value.map((item) => redactDeep(item));
      }
      if (value !== null && typeof value === "object") {
        return Object.fromEntries(
          Object.entries(value).map(([childKey, childValue]) => [
            childKey,
            redactDeep(childValue, childKey),
          ]),
        );
      }
      return value;
    }

    function flattenEntries(value, path = [], output = []) {
      const key = path.at(-1) ?? "";
      const sensitive = isSensitiveKey(key);
      const isObject = value !== null && typeof value === "object";
      const isPrimitiveArray = Array.isArray(value)
        && value.every((item) => item === null || typeof item !== "object");

      if (!isObject || sensitive || isPrimitiveArray || Object.keys(value).length === 0) {
        output.push({
          key,
          path: path.join("."),
          sensitive,
          value,
        });
        return output;
      }

      for (const [childKey, childValue] of Object.entries(value)) {
        flattenEntries(childValue, [...path, childKey], output);
      }
      return output;
    }

    function categorizeEntry(entry) {
      const key = entry.key.toLowerCase();
      const path = entry.path.toLowerCase();

      if (["iat", "nbf", "exp"].includes(key)) {
        return "time";
      }
      if (key === "scp" || key.includes("scope") || key.includes("permission")) {
        return "permissions";
      }
      if (path.startsWith("header.") || ["iss", "aud", "client_id"].includes(key)) {
        return "security";
      }
      if (["amr", "email_verified", "is_signup", "pwd_auth_time", "sl"].includes(key)
        || path.includes(".auth.")) {
        return "authentication";
      }
      if (entry.sensitive
        || path.includes("profile")
        || key.includes("plan")
        || key.includes("residency")
        || key.includes("user")
        || key.includes("account")
        || key.includes("organization")
        || key.includes("workspace")) {
        return "identity";
      }
      return "other";
    }

    function buildSections(header, payload) {
      const entries = [
        ...flattenEntries(header, ["header"]),
```

```js
    function createRevealRegistry({
      setTimeoutFn = setTimeout,
      clearTimeoutFn = clearTimeout,
      delay = 10_000,
    } = {}) {
      const timers = new Map();

      function hide(key) {
        if (!timers.has(key)) {
          return;
        }
        clearTimeoutFn(timers.get(key));
        timers.delete(key);
      }

      function show(key, onHide) {
        hide(key);
        const timerId = setTimeoutFn(() => {
          timers.delete(key);
          onHide();
        }, delay);
        timers.set(key, timerId);
      }

      function clear() {
        for (const timerId of timers.values()) {
          clearTimeoutFn(timerId);
        }
        timers.clear();
      }

      return Object.freeze({ clear, hide, show });
    }

    async function copyText(text, {
      navigatorRef = typeof navigator === "undefined" ? null : navigator,
      documentRef = typeof document === "undefined" ? null : document,
    } = {}) {
      if (navigatorRef?.clipboard?.writeText) {
        try {
          await navigatorRef.clipboard.writeText(text);
          return "clipboard";
        } catch {
          // Continue to the local textarea fallback.
        }
      }

      if (!documentRef?.body || typeof documentRef.execCommand !== "function") {
        throw new Error("当前浏览器不允许自动复制，请手动选择脱敏 JSON。");
      }

      const textarea = documentRef.createElement("textarea");
      textarea.value = text;
      textarea.setAttribute("readonly", "");
      textarea.style.position = "fixed";
      textarea.style.opacity = "0";
      documentRef.body.appendChild(textarea);
      textarea.select();
      const copied = documentRef.execCommand("copy");
      textarea.remove();

      if (!copied) {
        throw new Error("当前浏览器不允许自动复制，请手动选择脱敏 JSON。");
      }
      return "fallback";
    }

    function formatDisplayValue(value) {
      if (typeof value === "string") {
        return value;
      }
      return JSON.stringify(value, null, 2);
```

```js
          card.dataset.state = state;
        }
        card.append(
          makeElement("span", "summary-label", label),
          makeElement("strong", "summary-value", value),
        );
        statusSummary.appendChild(card);
      }

      function renderSummary(result) {
        const { claims } = result.time;
        addSummary("JWT 结构", "三段式 · Signature 未验证");
        addSummary("签名算法", String(result.header.alg ?? "未提供"));
        addSummary("签发时间", claims.iat.valid ? claims.iat.beijing : "未提供或格式错误");
        addSummary("到期时间", claims.exp.valid ? claims.exp.beijing : "未提供或格式错误");
        addSummary("时间状态", result.time.label, result.time.code);
      }

      function renderWarnings(warnings) {
        warningList.replaceChildren();
        for (const warning of warnings) {
          const item = makeElement("div", "warning-item", `⚠ ${warning.message}`);
          warningList.appendChild(item);
        }
      }

      function createFieldRow(entry) {
        const row = makeElement("div", "field-row");
        const key = makeElement("div", "field-key", entry.path);
        const value = makeElement("pre", "field-value");
        row.append(key, value);

        if (!entry.sensitive) {
          value.textContent = formatDisplayValue(entry.value);
          return row;
        }

        let revealed = false;
        const button = makeElement("button", "reveal-button", "显示 10 秒");
        button.type = "button";
        button.setAttribute("aria-label", `临时显示 ${entry.path}`);

        function conceal() {
          revealed = false;
          value.textContent = "••••••••";
          button.textContent = "显示 10 秒";
          button.setAttribute("aria-pressed", "false");
        }

        conceal();
        button.addEventListener("click", () => {
          if (revealed) {
            revealRegistry.hide(entry.path);
            conceal();
            return;
          }

          revealed = true;
          value.textContent = formatDisplayValue(entry.value);
          button.textContent = "立即隐藏";
          button.setAttribute("aria-pressed", "true");
          revealRegistry.show(entry.path, conceal);
        });
        row.appendChild(button);
        return row;
      }

      function renderSections(result) {
        sectionsRoot.replaceChildren();
        for (const section of buildSections(result.header, result.payload)) {
          if (section.entries.length === 0) {
            continue;
          }
          const card = makeElement("section", "claim-card");
          card.appendChild(makeElement("h3", "", section.title));
          const list = makeElement("div", "field-list");
          for (const entry of section.entries) {
            list.appendChild(createFieldRow(entry));
          }
          card.appendChild(list);
          sectionsRoot.appendChild(card);
        }
      }

      function handleParse() {
        setError();
        resetResult();

        try {
          const result = parseJwt(input.value);
          input.value = "";
          redactedResult = {
            header: redactDeep(result.header),
            payload: redactDeep(result.payload),
            signature: result.signature,
          };
          renderSummary(result);
          renderWarnings(result.warnings);
          renderSections(result);
          rawJson.textContent = JSON.stringify(redactedResult, null, 2);
          resultArea.hidden = false;
          resultArea.focus?.();
        } catch (error) {
          const message = error instanceof ParserError
            ? error.message
            : "解析失败，请确认输入是完整的三段式 JWT。";
          setError(message);
        }
      }

      parseButton.addEventListener("click", handleParse);
      clearButton.addEventListener("click", () => clearAll());
      copyButton.addEventListener("click", async () => {
        if (!redactedResult) {
          return;
        }
        const text = JSON.stringify(redactedResult, null, 2);
        try {
          await copyText(text, { documentRef, navigatorRef: globalThis.navigator });
          copyStatus.textContent = "已复制脱敏 JSON";
        } catch (error) {
          copyStatus.textContent = error.message;
          rawJson.focus();
          const selection = globalThis.getSelection?.();
          if (selection) {
            const range = documentRef.createRange();
            range.selectNodeContents(rawJson);
            selection.removeAllRanges();
            selection.addRange(range);
          }
        }
      });
      documentRef.addEventListener("keydown", (event) => {
        if (event.key === "Escape") {
          clearAll();
        }
      });
    }
```

# AT 本地解析器

一个只在浏览器本地解码单个 JWT 形式 AT 的静态工具。它展示 Header、Payload、时间状态、权限和脱敏 JSON。

```js
import test from "node:test";
import assert from "node:assert/strict";
import { execFileSync } from "node:child_process";
import { readFileSync } from "node:fs";

test("tracked project files contain no token-shaped JWT secrets", () => {
  const output = execFileSync("git", ["ls-files", "-z"], { encoding: "utf8" });
  const files = output.split("\0").filter(Boolean);
  const tokenPattern = /eyJ[A-Za-z0-9_-]{20,}\.[A-Za-z0-9_-]{20,}\.[A-Za-z0-9_-]{20,}/gu;
  const findings = [];

  for (const file of files) {
    if (!/\.(?:html|mjs|js|json|md|txt)$/iu.test(file)) {
      continue;
    }
    const content = readFileSync(file, "utf8");
    if (tokenPattern.test(content)) {
      findings.push(file);
    }
    tokenPattern.lastIndex = 0;
  }

  assert.deepEqual(findings, []);
});
```
